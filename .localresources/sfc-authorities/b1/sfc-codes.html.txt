     (function ($) {
         function getParameterByName(name, url) {
             if (!url) url = window.location.href;
             name = name.replace(/[\[\]]/g, "\\$&");
             var regex = new RegExp("[?&]" + name + "(=([^&#]*)|&|#|$)"),
                 results = regex.exec(url);
             if (!results) return null;
             if (!results[2]) return '';
             return decodeURIComponent(results[2].replace(/\+/g, " "));
         }

         $.fn.delayKeyup = function (callback, ms) {
             var timer = 0;
             var el = $(this);
             $(this).on('input propertychange', function () {
                 clearTimeout(timer);
                 timer = setTimeout(function () {
                     callback(el)
                 }, ms);
             });
             return $(this);
         };

         var lang = 'en';
         var idsForKeywords = [], idsForTopics = [];
         var hasKeywordsFilter = false, hasTopicsFilter = false;
         var $select = $('form.alert-list-search-panel :input[name="alert-list-display-option"]'),
             $input = $('form.alert-list-search-panel :input[name="search"]'),
             $rows = $('table tr[data-code-guideline-id]');

         var index = elasticlunr(function () {
             this.addField('title');
             this.addField('topics');
             this.setRef('id');
         });

         function buildIndex() {
             $.each($rows, function (i, tr) {
                 index.addDoc({
                     "id": $(tr).attr("data-code-guideline-id"),
                     "topics": $(tr).attr("data-code-guideline-topics").trim(),
                     "title": $(tr).children('td').first().text().trim()
                 });
             });
         }

         function populateIdsForKeywords() {
             var keyword = $input.val();
             keyword = $.trim(keyword);
             if (keyword && keyword.length > 0) {
                 keyword = keyword.toUpperCase();
                 var keywords = keyword.split(" ");
                 if (keyword.length > 2) {
                     var firstChar = keyword[0];
                     var lastChar = keyword[keyword.length - 1];
                     if (firstChar == '"' && lastChar == '"') {
                         keyword = keyword.replace(/"/g, '');
                         keywords = [keyword];
                     }
                 }

                 hasKeywordsFilter = true;
                 var ret = [];
                 for (var id in index.documentStore.docs) {
                     var doc = index.documentStore.docs[id];
                     for (var i = 0; i < keywords.length; i++) {
                         var kwd = keywords[i];
                         if (doc.title.toUpperCase().indexOf(kwd) !== -1 || doc.topics.toUpperCase().indexOf(kwd) !== -1) {
                             ret.push(doc);
                         }
                     }
                 }
                 for (var k in ret) {
                     var docRef = ret[k]['id'];
                     if (typeof docRef !== 'undefined') {
                         idsForKeywords.push(docRef);
                     }
                 }
             } else {
                 hasKeywordsFilter = false;
                 idsForKeywords = [];
             }
         }

         function populateIdsForTopics() {
             var topic = $select.val();
             topic = $.trim(topic);
             if (topic && topic.length > 0 && topic != 0) {
                 hasTopicsFilter = true;
                 var ret = [];
                 if (lang == "en") {
                     ret = index.search({
                         topics: topic
                     });
                 } else {
                     for (var id in index.documentStore.docs) {
                         var doc = index.documentStore.docs[id];
                         if (doc.topics.indexOf(topic) !== -1) {
                             ret.push({ ref: id });
                         }
                     }
                 }
                 for (var k in ret) {
                     var docRef = ret[k]['ref'];
                     if (typeof docRef !== 'undefined') {
                         idsForTopics.push(docRef);
                     }
                 }
             } else {
                 hasTopicsFilter = false;
                 idsForTopics = [];
             }
         }

         function doSearch() {
             $rows.hide();
             var hasFilters = hasKeywordsFilter || hasTopicsFilter;
             var ids = [];
             if (idsForTopics.length > 0 && idsForKeywords.length > 0) {
                 ids = $.grep(idsForTopics, function (n) {
                     return idsForKeywords.indexOf(n) !== -1;
                 });
             } else if (idsForTopics.length > 0) {
                 ids = idsForTopics;
             } else if (idsForKeywords.length > 0) {
                 ids = idsForKeywords;
             } else if (!hasFilters) {
                 $rows.show();
             }
             for (var x in ids) {
                 $rows.filter("[data-code-guideline-id=" + ids[x] + "]").show();
             }
         }

         buildIndex();

         $('form.alert-list-search-panel').on("submit", function (e) { e.preventDefault() });

         $select.on('change', function () {
             idsForTopics = [];
             populateIdsForTopics();
             doSearch();
         });

         $input.delayKeyup(function () {
             idsForKeywords = [];
             populateIdsForKeywords();
             doSearch();
         }, 500);

         var topic = getParameterByName('topic');
         var rule = getParameterByName('rule');
         if (topic && topic.length) {
             $select.val(topic);
         } else {
             topic = '';
         }

         if (rule && rule.length) {
             $input.val('"' + rule + '"');
         } else {
             rule = '';
         }

         if (rule.length || topic.length) {
             populateIdsForKeywords();
             populateIdsForTopics();
             doSearch();
         }

         $("#alert-list-reset-btn").click(function (e) {
             e.preventDefault();
             $input.val("");
             populateIdsForKeywords();
             $select.val("0").trigger('change');
             return false;
         });
     })(jQuery);
